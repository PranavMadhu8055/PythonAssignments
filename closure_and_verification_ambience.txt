class ClosureandVerificationAmbience extends StatefulWidget {
  const ClosureandVerificationAmbience({
    super.key,
    this.width,
    this.height,
    this.baseurl,
    this.branchId,
    this.empCode,

  final double? width;
  final double? height;

  State<ClosureandVerificationAmbience> createState() =>

class _ClosureandVerificationAmbienceState
    extends State<ClosureandVerificationAmbience> {
  final DateFormat _dateFormat = DateFormat('dd-MM-yyyy');
  List<String> _auditIds = [];
  String? _selectedAuditId;
  List<_AuditRow> _rows = [];

  bool _isLoadingList = false;
  bool _isLoadingDetails = false;
  bool _isSubmitting = false;
  String? _errorMessage;

  // Per-row inputs (same index as _rows)
  // status: 1 = Satisfactory closed, 2 = Wrong Rectification
  List<TextEditingController> _remarkControllers = [];
  List<int> _statuses = [];

  String get _baseUrl => (widget.baseurl ?? '').replaceAll(RegExp(r'/$'), '');

  void _disposeRowControllers() {
    for (final c in _remarkControllers) {
      c.dispose();
    }
    _remarkControllers = [];
    _statuses = [];
  }

  @override
  void dispose() {
    _disposeRowControllers();
    super.dispose();
  }

  @override
  void initState() {
    super.initState();
    _loadAuditIds();
  }

  // On load: fetch Audit IDs for the branch
  Future<void> _loadAuditIds() async {
    final branchId = widget.branchId?.trim();
    if (_baseUrl.isEmpty || branchId == null || branchId.isEmpty) {
      setState(() => _errorMessage = 'Branch and API URL are required.');
      return;
    }
    setState(() {
      _isLoadingList = true;
      _errorMessage = null;
    });
    try {
      final response = await http.get(Uri.parse(
        '$_baseUrl/AuditInspectionApi/api/GetInspectionData/'
        'Audit_id_ambience/$branchId/1',
      ));
      if (response.statusCode < 200 || response.statusCode >= 300) {
        throw Exception('Unable to load audit ids.');
      }
      final result = _resultList(response.body);
      if (!mounted) return;

      // API returns duplicate AUDIT_IDs, so de-duplicate (keeps order)
      final ids = <String>[];
      for (final row in result) {
        if (row is Map<String, dynamic>) {
          final id = '${row['AUDIT_ID'] ?? ''}'.trim();
          if (id.isNotEmpty && !ids.contains(id)) ids.add(id);
        }
      }
      setState(() {
        _auditIds = ids;
        _isLoadingList = false;
      });
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _isLoadingList = false;
        _errorMessage = 'Unable to load audit id list.';
      });
    }
  }

  // On select: fetch the datatable for the chosen Audit ID
  Future<void> _loadDetails(String auditId) async {
    if (_baseUrl.isEmpty) return;
    setState(() {
      _disposeRowControllers();
      _selectedAuditId = auditId;
      _rows = [];
      _isLoadingDetails = true;
      _errorMessage = null;
    });
    try {
      final response = await http.get(Uri.parse(
        '$_baseUrl/AuditInspectionApi/api/GetInspectionData/'
        'Audit_id_Datatable/$auditId/1',
      ));
      if (response.statusCode < 200 || response.statusCode >= 300) {
        throw Exception('Unable to load audit data.');
      }
      final rows = _resultList(response.body)
          .whereType<Map<String, dynamic>>()
          .map(_AuditRow.fromJson)
          .toList();
      if (!mounted) return;
      setState(() {
        _rows = rows;
        _remarkControllers =
            List.generate(rows.length, (_) => TextEditingController());
        _statuses = List.filled(rows.length, 1);
        _isLoadingDetails = false;
      });
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _isLoadingDetails = false;
        _errorMessage = 'Unable to load details for the selected audit id.';
      });
    }
  }

  List<dynamic> _resultList(String body) {
    final decoded = jsonDecode(body);
    return decoded is Map<String, dynamic> && decoded['result'] is List
        ? decoded['result'] as List<dynamic>
        : <dynamic>[];
  }

  Future<void> _showAlert(String message) async {
    if (!mounted) return;
    await showDialog<void>(
      context: context,
      builder: (context) => AlertDialog(
        title: const Text('Alert'),
        content: Text(message),
        actions: [
          TextButton(
            onPressed: () => Navigator.of(context).pop(),
            child: const Text('OK'),
          ),
        ],
      ),
    );
  }

  Future<void> _submitClosure(String flag) async {
    final auditId = _selectedAuditId;
    final branchId = widget.branchId?.trim();
    final empCode = widget.empCode?.trim();
    final currentDate = _dateFormat.format(DateTime.now());
    if (auditId == null || branchId == null || branchId.isEmpty) {
      await _showAlert('Please select an audit id first.');
      return;
    }
    if (_rows.isEmpty) {
      await _showAlert('No audit rows to verify.');
      return;
    }
    // every row needs its own remarks
    for (var i = 0; i < _rows.length; i++) {
      if (_remarkControllers[i].text.trim().isEmpty) {
        await _showAlert('Please enter remarks for row ${i + 1} '
            '(${_rows[i].category}).');
        return;
      }
    }
    if (empCode == null || empCode.isEmpty) {
      setState(() => _errorMessage = 'Employee code is required.');
      return;
    }
    setState(() {
      _isSubmitting = true;
      _errorMessage = null;
    });
    try {
      // One request per row. Category is sent so the backend can tell
      // the rows apart. Confirm this pageValue format with the backend team.
      String responseMessage = 'Successfully updated status.';
      for (var i = 0; i < _rows.length; i++) {
        final remarks = _remarkControllers[i].text.trim();
        final response = await http.post(
          Uri.parse('$_baseUrl/AuditInspectionApi/api/PostInseptionCtl'),
          headers: {'Content-Type': 'application/json'},
          body: jsonEncode({
            'flag': 'closure_and_verify_ambience',
            'pageValue':
                'approve*$branchId~$auditId~$empCode~${_statuses[i]}~$remarks~$currentDate~${_rows[i].category}',
            'paraValue': '1',
          }),
        );
        if (response.statusCode < 200 || response.statusCode >= 300) {
          throw Exception('Unable to submit row ${i + 1}.');
        }
        final decoded = jsonDecode(response.body);
        if (decoded is List &&
            decoded.isNotEmpty &&
            decoded.first is Map<String, dynamic>) {
          final res = '${(decoded.first as Map<String, dynamic>)['RES'] ?? ''}';
          if (res.isNotEmpty) responseMessage = res;
        }
      }
      if (!mounted) return;
      setState(() {
        _isSubmitting = false;
        _errorMessage = null;
        _disposeRowControllers();
        _selectedAuditId = null;
        _rows = [];
      });
      await showDialog<void>(
        context: context,
        builder: (context) => AlertDialog(
          title: Text(responseMessage),
          actions: [
            TextButton(
              onPressed: () => Navigator.of(context).pop(),
              child: const Text('OK'),
            ),
          ],
        ),
      );
      _loadAuditIds();
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _isSubmitting = false;
        _errorMessage = 'Unable to submit the closure action.';
      });
    }
  }

class _FormDropdown<T> extends StatelessWidget {
  final String label;
  final double width;
  final T? value;
  final List<T> items;
  final void Function(T?)? onChanged;
  final String Function(T) displayText;
  final String hintText;
  const _FormDropdown({
    Key? key,
    required this.label,
    required this.width,
    required this.value,
    required this.items,
    required this.onChanged,
    required this.displayText,
    this.hintText = 'Select...',
  }) : super(key: key);
  @override
  Widget build(BuildContext context) {
    return SizedBox(
      width: width,
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Text(
            label,
            style: const TextStyle(
              fontSize: 11,
              fontWeight: FontWeight.w600,
              color: Color(0xFF4A5568),
            ),
          ),
          const SizedBox(height: 8),
          Container(
            height: 36,
            padding: const EdgeInsets.symmetric(horizontal: 10),
            decoration: BoxDecoration(
              color: Colors.white,
              borderRadius: BorderRadius.circular(6),
              border: Border.all(color: const Color(0xFFCBD5E0)),
              boxShadow: [
                BoxShadow(
                  color: Colors.black.withOpacity(0.02),
                  blurRadius: 4,
                  offset: const Offset(0, 2),
                ),
              ],
            ),
            child: DropdownButtonHideUnderline(
              child: DropdownButton<T>(
                hint: Text(
                  hintText,
                  style: const TextStyle(fontSize: 12, color: Colors.grey),
                ),
                value: value,
                isExpanded: true,
                icon:
                    const Icon(Icons.arrow_drop_down, color: Color(0xFF4A5568)),
                items: items.map((item) {
                  return DropdownMenuItem<T>(
                    value: item,
                    child: Text(
                      displayText(item),
                      style: const TextStyle(fontSize: 12),
                      overflow: TextOverflow.ellipsis,
                    ),
                  );
                }).toList(),
                onChanged: onChanged,
                style: const TextStyle(
                  fontSize: 12,
                  color: Color(0xFF1A202C),
                ),
              ),
            ),
          ),
        ],
      ),
    );
  }
}

// --- Data Models
class _AuditRow {
  const _AuditRow({
    required this.category,
    required this.question,
    required this.auditorRemark,
    required this.bhAnswer,
  });
  final String category;
  final String question;
  final String auditorRemark;
  final String bhAnswer;
  factory _AuditRow.fromJson(Map<String, dynamic> json) => _AuditRow(
        category: '${json['CATEGORY'] ?? ''}',
        question: '${json['QUESTION'] ?? ''}',
        auditorRemark: '${json['AUDITOR_REMARK'] ?? ''}',
        bhAnswer: '${json['BH_ANSWER'] ?? ''}',
      );
}

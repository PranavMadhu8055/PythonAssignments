  Future<void> _loadPledges() async {
    final branchId = widget.branchId?.trim();
    if (_baseUrl.isEmpty || branchId == null || branchId.isEmpty) {
      setState(() => _errorMessage = 'Branch and API URL are required.');
      return;
    }
    setState(() {
      _isLoadingPledges = true;
      _errorMessage = null;
    });
    try {
      final response = await http.get(Uri.parse(
        '$_baseUrl/AuditInspectionApi/api/GetInspectionData/'
        'IrrCodeDropdown/$branchId/1',
      ));
      if (response.statusCode < 200 || response.statusCode >= 300) {
        throw Exception('Unable to load pledge numbers.');
      }
      final result = _resultList(response.body);
      if (!mounted) return;
      setState(() {
        _pledges = result
            .whereType<Map<String, dynamic>>()
            .map(_PledgeOption.fromJson)
            .where((pledge) => pledge.pledgeNo.isNotEmpty)
            .toList();
        _isLoadingPledges = false;
      });
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _isLoadingPledges = false;
        _errorMessage = 'Unable to load pledge numbers.';
      });
    }
  }

  Future<void> _loadPledgeData(_PledgeOption pledge) async {
    final branchId = widget.branchId?.trim();
    if (_baseUrl.isEmpty || branchId == null || branchId.isEmpty) return;
    setState(() {
      _selectedPledge = pledge;
      _pledgeDetails = null;
      _transactions = [];
      _isLoadingDetails = true;
      _errorMessage = null;
    });
    try {
      final detailsUri = Uri.parse(
        '$_baseUrl/AuditInspectionApi/api/GetInspectionData/'
        'IrrCodeDropdownDetails/$branchId*${pledge.irrCode}/1',
      );
      final transactionsUri = Uri.parse(
        '$_baseUrl/AuditInspectionApi/api/GetInspectionData/'
        'cus_tranDetails/${pledge.pledgeNo}/1',
      );
      final settlementTypeUri = Uri.parse(
        '$_baseUrl/AuditInspectionApi/api/GetInspectionData/'
        'settlementTypes/${pledge.pledgeNo}/1',
      );
      final getLossUri = Uri.parse(
        '$_baseUrl/AuditInspectionApi/api/GetInspectionData/'
        'lossAmountDetails/${pledge.pledgeNo}*${pledge.irrCode}/1',
      );
      final responses = await Future.wait([
        http.get(detailsUri),
        http.get(transactionsUri),
        http.get(settlementTypeUri),
        http.get(getLossUri),
      ]);
      for (final response in responses) {
        if (response.statusCode < 200 || response.statusCode >= 300) {
          throw Exception('Unable to load pledge data.');
        }
      }
      final detailRows = _resultList(responses[0].body);
      final transactionRows = _resultList(responses[1].body);
      final settlementTypeRows = _resultList(responses[2].body);
      final lossRows = _resultList(responses[3].body);
      double? lossValue;
      if (lossRows.isNotEmpty && lossRows.first is Map<String, dynamic>) {
        lossValue = (lossRows.first as Map<String, dynamic>)['RES'];
      }
      String? settlementTypesRow;
      if (settlementTypeRows.isNotEmpty &&
          settlementTypeRows.first is Map<String, dynamic>) {
        settlementTypesRow =
            (settlementTypeRows.first as Map<String, dynamic>)['RES'];
      }
      if (!mounted) return;
      setState(() {
        if (detailRows.isNotEmpty && detailRows.first is Map<String, dynamic>) {
          final detail =
              _PledgeDetails.fromJson(detailRows.first as Map<String, dynamic>);
          _pledgeDetails = _PledgeDetails(
            pledgeDate: detail.pledgeDate,
            irrcode: detail.irrcode,
            pledgeValue: detail.pledgeValue,
            irregularityDate: detail.irregularityDate,
            irregularityType: detail.irregularityType,
            settlementDate: detail.settlementDate,
            settlementType: settlementTypesRow,
            lossAmount: detail.lossAmount,
            lossWt: lossValue?.toString(),
          );
        } else {
          _pledgeDetails = null;
        }
        _transactions = transactionRows
            .whereType<Map<String, dynamic>>()
            .map(_Transaction.fromJson)
            .toList();
        _isLoadingDetails = false;
      });
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _isLoadingDetails = false;
        _errorMessage = 'Unable to load details for the selected pledge.';
      });
    }
  }

  // Table Select Data API Fetch
  Future<void> _updateSelectDataOptions(int index, int classification) async {
    final row = _irregularityRows[index];
    setState(() {
      row.auditorClassification = classification;
      row.selectData = null; // Reset selection on change
      row.selectDataName = '';
      row.isFetchingOptions = true;
      row.isSelectDataReadOnly = false; // Reset read-only status
    });
    String param = '';
    if (classification == 1 || classification == 3) {
      param = '60*0';
    } else if (classification == 2) {
      param = '8*${row.id}';
    }
    if (param.isNotEmpty) {
      try {
        final response = await http.get(Uri.parse(
            '$_baseUrl/AuditInspectionApi/api/GetInspectionData/PROC_VERIFY_AND_CLOSURE/$param/1'));
        if (response.statusCode == 200) {
          final decoded = jsonDecode(response.body);
          if (decoded != null && decoded['result'] is List) {
            final List<dynamic> result = decoded['result'];
            final List<Map<String, dynamic>> fetchedOptions = result
                .map((e) {
                  dynamic rawId = e['0'];
                  int id = (rawId is num)
                      ? rawId.toInt()
                      : (int.tryParse(rawId?.toString() ?? '0') ?? 0);
                  return {
                    'id': id,
                    'name': e["'----------SELECT----------'"]?.toString() ?? ''
                  };
                })
                .where((e) => e['id'] != 0)
                .toList(); // Filter out '0' default select option
            if (mounted) {
              setState(() {
                row.selectDataOptions = fetchedOptions;

                // --- AUTO SELECT & READ ONLY LOGIC START --
                if (classification == 1) {
                  try {
                    final autoOption = fetchedOptions.firstWhere((e) =>
                        (e['name'] as String)
                            .toLowerCase()
                            .contains('verified & ok') ||
                        (e['name'] as String)
                            .toLowerCase()
                            .contains('verified'));
                    row.selectData = autoOption['id'] as int;
                    row.selectDataName = autoOption['name'] as String;
                    row.isSelectDataReadOnly = true; // Lock dropdown
                  } catch (_) {}
                } else if (classification == 3) {
                  try {
                    final autoOption = fetchedOptions.firstWhere((e) =>
                        (e['name'] as String)
                            .toLowerCase()
                            .contains('not applicable') ||
                        (e['name'] as String)
                            .toLowerCase()
                            .contains('applicable'));
                    row.selectData = autoOption['id'] as int;
                    row.selectDataName = autoOption['name'] as String;
                    row.isSelectDataReadOnly = true; // Lock dropdown
                  } catch (_) {}
                }
                // --- AUTO SELECT & READ ONLY LOGIC END --
              });
            }
          }
        }
      } catch (e) {
        debugPrint('Error fetching dropdown options: $e');
      }
    } else {
      if (mounted) {
        setState(() {
          row.selectDataOptions = [];
        });
      }
    }
    if (mounted) {
      setState(() {
        row.isFetchingOptions = false;
      });
    }
  }

  List<dynamic> _resultList(String body) {
    final decoded = jsonDecode(body);
    return decoded is Map<String, dynamic> && decoded['result'] is List
        ? decoded['result'] as List<dynamic>
        : <dynamic>[];
  }

  String _formatDate(String? value) {
    if (value == null || value.isEmpty) return '-';
    try {
      return _dateFormat.format(DateTime.parse(value).toLocal());
    } catch (_) {
      return value;
    }
  }

  String _formatAmount(num? value) => value == null ? '-' : value.toString();

  Future<void> _submitClosure(String flag) async {
    final pledge = _selectedPledge;
    final branchId = widget.branchId?.trim();
    final empCode = widget.empCode?.trim();
    final remarks = _remarksController.text.trim();

    if (pledge == null || branchId == null || branchId.isEmpty) {
      await _showAlert('Please select a pledge first.');
      return;
    }
    // Check if table auditor classifications are selected (if applicable)
    bool tableIncomplete =
        _irregularityRows.any((row) => row.auditorClassification == null);
    if (tableIncomplete) {
      await _showAlert(
          'Please select AUDITOR CLASSIFICATION for all categories.');
      return;
    }
    if (remarks.isEmpty) {
      await _showAlert('Please enter remarks.');
      return;
    }
    if (empCode == null || empCode.isEmpty) {
      setState(() => _errorMessage = 'Employee code is required.');
      return;
    }
    setState(() {
      _isSubmitting = true;
      _errorMessage = null;
    });
    try {
      List<String> rowStrings = [];

      for (var row in _irregularityRows) {
        final id = row.id;

        final category = row.category.replaceAll(RegExp(r'[~!\*@]'), ' ');

        final rowRemarks = row.remarks.replaceAll(RegExp(r'[~!\*@]'), ' ');
        final spotRect = row.spotRectified ? '1' : '0';
        final auditorClassification = row.auditorClassification;
        final issuename = row.selectDataName;
        rowStrings.add(
            '${id}@${auditorClassification}@${category}@${issuename}@${rowRemarks}@${spotRect}');
      }

      final tableDataString = rowStrings.join('!');

      final pInData =
          '$branchId~${pledge.irrCode}~${pledge.pledgeNo}~$empCode~$selectedStatus~$remarks~$tableDataString';
      final response = await http.post(
        Uri.parse('$_baseUrl/AuditInspectionApi/api/PostInseptionCtl'),
        headers: {'Content-Type': 'application/json'},
        body: jsonEncode({
          'flag': 'closure_and_verify_post',
          'pageValue': 'approve*$pInData',
          'paraValue': '1',
        }),
      );
      if (response.statusCode < 200 || response.statusCode >= 300) {
        throw Exception('Unable to submit closure action.');
      }
      if (!mounted) return;
      final decodedResponse = jsonDecode(response.body);
      final responseMessage = decodedResponse is List &&
              decodedResponse.isNotEmpty &&
              decodedResponse.first is Map<String, dynamic>
          ? '${(decodedResponse.first as Map<String, dynamic>)['RES'] ?? ''}'
          : '';
      setState(() {
        _isSubmitting = false;
        _errorMessage = null;
        _remarksController.clear();
        _selectedPledge = null;
        _pledgeDetails = null;
        _transactions.clear();
        for (var row in _irregularityRows) {
          row.auditorClassification = null;
          row.selectData = null;
          row.remarks = '';
          row.spotRectified = false;
        }
      });
      await showDialog<void>(
        context: context,
        builder: (context) => AlertDialog(
          title: Text(
            responseMessage.isNotEmpty
                ? responseMessage
                : 'Error: No response message received.',
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.of(context).pop(),
              child: const Text('OK'),
            ),
          ],
        ),
      );
      _loadPledges();
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _isSubmitting = false;
        _errorMessage = 'Unable to submit the closure action.';
      });
    }
  }

class _GridDropdown<T> extends StatelessWidget {
  final T? value;
  final List<T> items;
  final void Function(T?)? onChanged;
  final String Function(T) displayText;
  final String hintText;
  final bool readOnly; // Added flag for read-only styling

  const _GridDropdown({
    required this.value,
    required this.items,
    required this.onChanged,
    required this.displayText,
    this.hintText = 'Select...',
    this.readOnly = false,
  });
  @override
  Widget build(BuildContext context) {
    return Container(
      height: 36,
      padding: const EdgeInsets.symmetric(horizontal: 10),
      decoration: BoxDecoration(
        color: readOnly
            ? const Color(0xFFF7F8FA)
            : Colors.white, // Gray out if readOnly
        borderRadius: BorderRadius.circular(4),
        border: Border.all(color: const Color(0xFFCBD5E0)),
      ),
      child: DropdownButtonHideUnderline(
        child: DropdownButton<T>(
          isExpanded: true,
          value: items.contains(value) ? value : null,
          hint: Text(hintText,
              style: const TextStyle(fontSize: 11, color: Colors.grey)),
          icon: Icon(Icons.keyboard_arrow_down,
              size: 18,
              color: readOnly ? Colors.grey : const Color(0xFF4A5568)),
          items: items.map((e) {
            return DropdownMenuItem(
              value: e,
              child: Text(
                displayText(e),
                style: TextStyle(
                    fontSize: 11,
                    color: readOnly
                        ? Colors.grey.shade700
                        : const Color(0xFF1A202C)),
                overflow: TextOverflow.ellipsis,
              ),
            );
          }).toList(),
          onChanged: readOnly
              ? null
              : onChanged, // Disable selection callback if readOnly
        ),
      ),
    );
  }
}

class _PledgeOption {
  const _PledgeOption({required this.irrCode, required this.pledgeNo});
  final String irrCode;
  final String pledgeNo;
  factory _PledgeOption.fromJson(Map<String, dynamic> json) {
    final rawIrrCode = '${json['IRR_CODE'] ?? ''}';
    final actualIrrCode =
        rawIrrCode.contains('~') ? rawIrrCode.split('~')[0] : rawIrrCode;
    return _PledgeOption(
      irrCode: actualIrrCode,
      pledgeNo: '${json['PLEDGE_NO'] ?? ''}',
    );
  }
  @override
  bool operator ==(Object other) =>
      other is _PledgeOption &&
      other.irrCode == irrCode &&
      other.pledgeNo == pledgeNo;
  @override
  int get hashCode => Object.hash(irrCode, pledgeNo);
}

class _Transaction {
  const _Transaction({
    required this.pledgeNo,
    required this.transactionDate,
    required this.amount,
    required this.type,
    required this.description,
  });
  final String pledgeNo;
  final String? transactionDate;
  final num? amount;
  final String type;
  final String description;
  factory _Transaction.fromJson(Map<String, dynamic> json) => _Transaction(
        pledgeNo: '${json['PLEDGE_NO'] ?? ''}',
        transactionDate: json['TRA_DT']?.toString(),
        amount: _asNum(json['AMOUNT']),
        type: '${json['TYPE'] ?? ''}'.toUpperCase(),
        description: '${json['DESCR'] ?? ''}',
      );
}

num? _asNum(dynamic value) => value is num ? value : num.tryParse('$value');

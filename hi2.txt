  Future<void> _loadIrregularities() async {
    final branchId = widget.branchId?.trim();
    if (_baseUrl.isEmpty || branchId == null || branchId.isEmpty) {
      setState(() => _errorMessage = 'Branch and API URL are required.');
      return;
    }
    setState(() {
      _isLoadingList = true;
      _errorMessage = null;
    });
    try {
      final response = await http.get(Uri.parse(
        '$_baseUrl/AuditInspectionApi/api/GetInspectionData/'
        'AmbienceIrrDropdown/$branchId/1', // Adjusted endpoint for Ambience
      ));
      if (response.statusCode < 200 || response.statusCode >= 300) {
        throw Exception('Unable to load irregularities.');
      }
      final result = _resultList(response.body);
      if (!mounted) return;
      setState(() {
        // _irregularities = result
        //     .whereType()
        //     .map(_IrrOption.fromJson)
        //     .where((irr) => irr.refNo.isNotEmpty)
        //     .toList();
        _isLoadingList = false;
      });
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _isLoadingList = false;
        _errorMessage = 'Unable to load irregularities list.';
      });
    }
  }

  Future<void> _loadDetails(_IrrOption irr) async {
    final branchId = widget.branchId?.trim();
    if (_baseUrl.isEmpty || branchId == null || branchId.isEmpty) return;
    setState(() {
      _selectedIrregularity = irr;
      _details = null;
      _isLoadingDetails = true;
      _errorMessage = null;
    });
    try {
      final detailsUri = Uri.parse(
        '$_baseUrl/AuditInspectionApi/api/GetInspectionData/'
        'AmbienceIrrDropdownDetails/$branchId*${irr.irrCode}/1',
      );

      final response = await http.get(detailsUri);

      if (response.statusCode < 200 || response.statusCode >= 300) {
        throw Exception('Unable to load irregularity data.');
      }

      final detailRows = _resultList(response.body);
      if (!mounted) return;
      setState(() {
        if (detailRows.isNotEmpty && detailRows.first is Map<String, dynamic>) {
          _details = _AmbienceDetails.fromJson(
              detailRows.first as Map<String, dynamic>);
        } else {
          _details = null;
        }
        _isLoadingDetails = false;
      });
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _isLoadingDetails = false;
        _errorMessage = 'Unable to load details for the selected irregularity.';
      });
    }
  }

  List<dynamic> _resultList(String body) {
    final decoded = jsonDecode(body);
    return decoded is Map<String, dynamic> && decoded['result'] is List
        ? decoded['result'] as List<dynamic>
        : <dynamic>[];
  }

  String _formatDate(String? value) {
    if (value == null || value.isEmpty) return '-';
    try {
      return _dateFormat.format(DateTime.parse(value).toLocal());
    } catch (_) {
      return value;
    }
  }

  Future<void> _showAlert(String message) async {
    if (!mounted) return;
    await showDialog<void>(
      context: context,
      builder: (context) => AlertDialog(
        title: const Text('Alert'),
        content: Text(message),
        actions: [
          TextButton(
            onPressed: () => Navigator.of(context).pop(),
            child: const Text('OK'),
          ),
        ],
      ),
    );
  }

  Future<void> _submitClosure(String flag) async {
    final irr = _selectedIrregularity;
    final branchId = widget.branchId?.trim();
    final empCode = widget.empCode?.trim();
    final remarks = _remarksController.text.trim();
    final currentDate = _dateFormat.format(DateTime.now());
    if (irr == null || branchId == null || branchId.isEmpty) {
      await _showAlert('Please select an irregularity first.');
      return;
    }
    if (remarks.isEmpty) {
      await _showAlert('Please enter remarks.');
      return;
    }
    if (empCode == null || empCode.isEmpty) {
      setState(() => _errorMessage = 'Employee code is required.');
      return;
    }
    setState(() {
      _isSubmitting = true;
      _errorMessage = null;
    });
    try {
      // Includes Validation Emp Code, Status, and Remarks
      final response = await http.post(
        Uri.parse('$_baseUrl/AuditInspectionApi/api/PostInseptionCtl'),
        headers: {'Content-Type': 'application/json'},
        body: jsonEncode({
          'flag': 'closure_and_verify_ambience',
          'pageValue':
              'approve*$branchId~${irr.irrCode}~${irr.refNo}~$empCode~$selectedStatus~$remarks~$currentDate',
          'paraValue': '1',
        }),
      );
      if (response.statusCode < 200 || response.statusCode >= 300) {
        throw Exception('Unable to submit closure action.');
      }
      if (!mounted) return;
      final decodedResponse = jsonDecode(response.body);
      final responseMessage = decodedResponse is List &&
              decodedResponse.isNotEmpty &&
              decodedResponse.first is Map<String, dynamic>
          ? '${(decodedResponse.first as Map<String, dynamic>)['RES'] ?? ''}'
          : 'Successfully updated status.';

      setState(() {
        _isSubmitting = false;
        _errorMessage = null;
        _remarksController.clear();
        _selectedIrregularity = null;
        _details = null;
      });
      await showDialog<void>(
        context: context,
        builder: (context) => AlertDialog(
          title: Text(responseMessage),
          actions: [
            TextButton(
              onPressed: () => Navigator.of(context).pop(),
              child: const Text('OK'),
            ),
          ],
        ),
      );
      _loadIrregularities();
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _isSubmitting = false;
        _errorMessage = 'Unable to submit the closure action.';
      });
    }
  }

class _IrrOption {
  const _IrrOption({required this.irrCode, required this.refNo});
  final String irrCode;
  final String refNo;
  factory _IrrOption.fromJson(Map<String, dynamic> json) {
    final rawIrrCode = '${json['IRR_CODE'] ?? ''}';
    final actualIrrCode =
        rawIrrCode.contains('~') ? rawIrrCode.split('~')[0] : rawIrrCode;
    return _IrrOption(
      irrCode: actualIrrCode,
      refNo: '${json['REF_NO'] ?? json['PLEDGE_NO'] ?? ''}',
    );
  }
  @override
  bool operator ==(Object other) =>
      other is _IrrOption && other.irrCode == irrCode && other.refNo == refNo;
  @override
  int get hashCode => Object.hash(irrCode, refNo);
}
